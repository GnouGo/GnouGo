sealed class SavedResponses(string[] scripts):ILLMClient,ILLMCapabilityResolver {
 public int Calls {get;private set;}
 public Task<int?> InputTokenAllowanceAsync(string? provider,string model,int outputTokens,CancellationToken ct)=>Task.FromResult<int?>(96000);
 public Task<bool?> SupportsStructuredOutputAsync(string? provider,string model,CancellationToken ct)=>Task.FromResult<bool?>(true);
 public Task<IReadOnlyList<string>?> SupportedReasoningLevelsAsync(string? provider,string model,CancellationToken ct)=>Task.FromResult<IReadOnlyList<string>?>([]);
 public Task<LLMResponse> CallAsync(LLMRequest request,CancellationToken ct){ if(Calls>=scripts.Length)throw new InvalidOperationException("No retained response; stop without dispatch."); return Task.FromResult(new LLMResponse{Json=new JsonObject{["script"]=scripts[Calls++]},Usage=new JsonObject{["input_tokens"]=0,["output_tokens"]=0,["total_tokens"]=0}});}
}
