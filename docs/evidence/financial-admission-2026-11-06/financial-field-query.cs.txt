 void Scan(System.Text.Json.Nodes.JsonNode? node,string path,string requestId) {
  if(node is System.Text.Json.Nodes.JsonObject obj)foreach(var pair in obj) {
   if(!System.Text.RegularExpressions.Regex.IsMatch(pair.Key,@"^[a-zA-Z0-9_.\[\]-]{1,96}$"))continue;
   var key=pair.Key.ToLowerInvariant();
   if(key is "input" or "output" or "choices" or "messages" or "content" or "text" or "parsed" or "json" or "arguments" or "instructions" or "schema" or "authorization" or "proxy-authorization" or "cookie" or "set-cookie" or "api-key" or "x-api-key")continue;
   var child=path+"."+pair.Key;
   if(System.Text.RegularExpressions.Regex.IsMatch(pair.Key,"cost|charge|bill|price|credit|premium|amount|currency|fee|usd|eur|monetary|balance|payment",System.Text.RegularExpressions.RegexOptions.IgnoreCase)) {
    paths[child]=paths.GetValueOrDefault(child)+1;
    decimal amount;
    if(pair.Value is System.Text.Json.Nodes.JsonValue v && (v.TryGetValue<decimal>(out amount)||v.TryGetValue<string>(out var str)&&decimal.TryParse(str,System.Globalization.NumberStyles.Number,System.Globalization.CultureInfo.InvariantCulture,out amount)))
     amounts.Add(new System.Text.Json.Nodes.JsonObject{["requestId"]=requestId,["path"]=child,["amount"]=amount.ToString("G29",System.Globalization.CultureInfo.InvariantCulture)});
   }
   Scan(pair.Value,child,requestId);
  }
  else if(node is System.Text.Json.Nodes.JsonArray a)foreach(var item in a)Scan(item,path+"[]",requestId);
 }
