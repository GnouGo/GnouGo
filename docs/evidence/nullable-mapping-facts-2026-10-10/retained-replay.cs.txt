sealed class Cache:IMappingArtifactStore{
 readonly Dictionary<(string,string),MappingArtifact> values=[];
 public Task<MappingArtifact?> ReadAsync(string tenant,string key,CancellationToken c)=>Task.FromResult(values.GetValueOrDefault((tenant,key)));
 public Task WriteAsync(string tenant,MappingArtifact a,CancellationToken c){values[(tenant,a.Key)]=a;return Task.CompletedTask;}
 public Task RemoveAsync(string tenant,string key,CancellationToken c){values.Remove((tenant,key));return Task.CompletedTask;}
}
