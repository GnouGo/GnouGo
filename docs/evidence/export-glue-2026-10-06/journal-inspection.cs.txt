async Task Visit(JsonNode? reference) {
    var r=reference!.AsArray(); if(r[0]!.GetValue<int>()==0)return;
    var hash=r[1]!.GetValue<string>(); if(!visited.Add(hash))return;
    var record=await records.GetAsync("flow-execution-blocks-v1",tenant,prefix+hash,author) ?? throw new Exception("Missing block");
    bytes+=Encoding.UTF8.GetByteCount(record.Value);var e=JsonNode.Parse(record.Value)!;
    if(e["tenantId"]!.ToString()!=tenant || e["runId"]!.ToString()!=runId)throw new Exception("Wrong owner");
    var b=e["body"]!;
    if(Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(b.ToJsonString())))!=hash)throw new Exception("Wrong block hash");
    if(b["kind"]!.ToString()=="object") foreach(var v in b["values"]!.AsArray()) await Visit(v![1]);
    else if(b["kind"]!.ToString() is "array" or "chunks") foreach(var v in b["values"]!.AsArray()) await Visit(v);
}
